// theme.ts
const derive = (primary) => {
  const { h, s, l } = primary
  return {
    accent: hsl(h + 40, s * 0.8, l),
    muted:  hsl(h, s * 0.3, 0.6),
    mode: luminance(primary) < 0.45
      ? "light" : "dark",
  }
